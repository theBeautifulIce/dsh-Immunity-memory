```protobuf
syntax = "proto3";

package deepseek.memory;

option go_package = "deepseek-memory/proto";

// ==================== 搜索服务 ====================

service SearchService {
  // 基础搜索（全文搜索）
  rpc Search(SearchRequest) returns (SearchResponse);
  
  // 关系图谱搜索
  rpc GraphSearch(GraphSearchRequest) returns (GraphSearchResponse);
  
  // 混合搜索（全文 + 图谱）
  rpc HybridSearch(HybridSearchRequest) returns (HybridSearchResponse);
  
  // 索引记忆（更新搜索索引）
  rpc IndexMemory(IndexRequest) returns (IndexResponse);
  
  // 删除索引
  rpc RemoveIndex(RemoveIndexRequest) returns (RemoveIndexResponse);
  
  // 批量索引
  rpc BatchIndex(BatchIndexRequest) returns (BatchIndexResponse);
  
  // 健康检查
  rpc HealthCheck(HealthCheckRequest) returns (HealthCheckResponse);
  
  // 获取统计信息
  rpc GetStats(StatsRequest) returns (StatsResponse);
}

// ==================== 消息定义 ====================

// 搜索请求
message SearchRequest {
  string query = 1;                    // 搜索关键词
  SearchFilters filters = 2;           // 过滤条件
  int32 limit = 3;                     // 返回数量限制（默认20）
  int32 offset = 4;                    // 分页偏移
  bool include_scores = 5;             // 是否返回评分
  SearchMode mode = 6;                 // 搜索模式
}

// 搜索模式
enum SearchMode {
  FULLTEXT = 0;      // 纯全文搜索
  SEMANTIC = 1;      // 纯语义搜索（需要向量）
  HYBRID = 2;        // 混合搜索（默认）
}

// 搜索过滤器
message SearchFilters {
  optional string layer = 1;           // 层级过滤（L2/L4/L5/L6）
  optional string type = 2;            // 类型过滤
  repeated string tags = 3;            // 标签过滤
  optional int64 start_time = 4;       // 时间范围起始（Unix 时间戳）
  optional int64 end_time = 5;         // 时间范围结束
  optional float min_importance = 6;   // 最小重要性
  optional string status = 7;          // 状态过滤（active/archived/deleted）
}

// 搜索响应
message SearchResponse {
  repeated string memory_ids = 1;      // 记忆 ID 列表
  map<string, float> scores = 2;       // ID → 评分映射
  int32 total_count = 3;               // 总命中数
  float search_time_ms = 4;            // 搜索耗时（毫秒）
}

// 关系图谱搜索请求
message GraphSearchRequest {
  string start_memory_id = 1;          // 起始记忆 ID
  int32 max_depth = 2;                 // 最大遍历深度（默认2）
  int32 max_results = 3;               // 最大返回数量（默认50）
  repeated string edge_types = 4;      // 边类型过滤（learned_from/related_to/supports）
  float min_weight = 5;                // 最小边权重（默认0.5）
  TraversalMode traversal_mode = 6;    // 遍历模式
}

// 遍历模式
enum TraversalMode {
  BFS = 0;           // 广度优先（默认）
  DFS = 1;           // 深度优先
  BIDIRECTIONAL = 2; // 双向搜索
}

// 关系图谱搜索响应
message GraphSearchResponse {
  repeated GraphNode nodes = 1;        // 节点列表
  repeated GraphEdge edges = 2;        // 边列表
  float search_time_ms = 3;            // 搜索耗时
}

// 图节点
message GraphNode {
  string memory_id = 1;
  int32 depth = 2;                     // 距离起始节点的深度
  float relevance = 3;                 // 相关性评分
}

// 图边
message GraphEdge {
  string source_id = 1;
  string target_id = 2;
  string edge_type = 3;
  float weight = 4;
}

message HybridSearchRequest {
  SearchRequest search_req = 1;        // 基础搜索请求
  GraphSearchRequest graph_req = 2;    // 图谱搜索请求（可选）
  HybridWeights weights = 3;           // 结果融合权重
}

// 混合搜索权重

  float graph_weight = 3;              // 图谱搜索权重（默认0.2）
}

// 混合搜索响应
message HybridSearchResponse {

  float search_time_ms = 3;
```

```python
from typing import List, Dict, Optional
import time

from ..proto import memory_pb2, memory_pb2_grpc


class SearchClient:
    """
    Go 搜索服务的 gRPC 客户端
    """
    
    def __init__(self, address: str):
        self.address = address
        self.channel = grpc.insecure_channel(address)
        self.stub = memory_pb2_grpc.SearchServiceStub(self.channel)
        
        # 等待服务就绪
        self._wait_for_ready()
    
    def _wait_for_ready(self, timeout: int = 10):
        """等待服务就绪"""
        start = time.time()
        while time.time() - start < timeout:
            try:
                self.health_check()
                return
            except grpc.RpcError:
                time.sleep(0.1)
        raise TimeoutError(f"等待 gRPC 服务就绪超时: {self.address}")
    
    def search(
        self,
        query: str,
        filters: Optional[Dict] = None,
        limit: int = 20,
        offset: int = 0,
        include_scores: bool = False
    ) -> List[str]:
        """
        基础搜索
        
        Returns:
            memory_id 列表
        """
        # 构建过滤器
        filter_pb = memory_pb2.SearchFilters()
        if filters:
            if 'layer' in filters:
                filter_pb.layer = filters['layer']
            if 'type' in filters:
                filter_pb.type = filters['type']
            if 'tags' in filters:
                filter_pb.tags.extend(filters['tags'])
            if 'min_importance' in filters:
                filter_pb.min_importance = filters['min_importance']
        
        # 构建请求
        request = memory_pb2.SearchRequest(
            query=query,
            filters=filter_pb,
            limit=limit,
            offset=offset,
            include_scores=include_scores,

        )

        try:
            response = self.stub.Search(request, timeout=10.0)
            
            if include_scores:
                return list(response.memory_ids), dict(response.scores)
            else:
                return list(response.memory_ids)
        
        except grpc.RpcError as e:
            raise RuntimeError(f"搜索失败: {e.details()}")

        self,
        start_memory_id: str,
        max_depth: int = 2,
        max_results: int = 50,

        min_weight: float = 0.5

        关系图谱搜索
```

## 2. Go 图谱遍历算法

### 2.1 图谱搜索服务实现

```go
// go-service/internal/graph/search.go
package graph

import (
    "container/heap"
    "context"
    "database/sql"
    "fmt"
    "sync"
)

// Node 图节点
type Node struct {
    MemoryID  string
    Depth     int
    Relevance float32
}

// Edge 图边
type Edge struct {
    SourceID string
    TargetID string
    EdgeType string
    Weight   float32
}

// Graph 图结构
type Graph struct {
    db    *sql.DB
    cache *sync.Map // 缓存已访问的节点
}

func NewGraph(db *sql.DB) *Graph {
    return &Graph{
        db:    db,
        cache: &sync.Map{},
    }
}

// BFSSearch 广度优先搜索
func (g *Graph) BFSSearch(
    ctx context.Context,
    startID string,
    maxDepth int,
    maxResults int,
    edgeTypes []string,
    minWeight float32,
) ([]Node, []Edge, error) {
    
    visited := make(map[string]bool)
    queue := []Node{{MemoryID: startID, Depth: 0, Relevance: 1.0}}
    
    var resultNodes []Node
    var resultEdges []Edge
    
    for len(queue) > 0 && len(resultNodes) < maxResults {
        // 出队
        current := queue[0]
        queue = queue[1:]
        
        // 检查是否已访问
        if visited[current.MemoryID] {
            continue
        }
        visited[current.MemoryID] = true
        
        // 添加到结果
        resultNodes = append(resultNodes, current)
        
        // 如果达到最大深度，停止扩展
        if current.Depth >= maxDepth {
            continue
        }
        
        // 查询邻居节点
        neighbors, edges, err := g.getNeighbors(ctx, current.MemoryID, edgeTypes, minWeight)
        if err != nil {
            return nil, nil, fmt.Errorf("查询邻居失败: %w", err)
        }
        
        // 将邻居加入队列
        for i, neighbor := range neighbors {
            if !visited[neighbor] {
                // 计算相关性（根据边权重衰减）
                relevance := current.Relevance * edges[i].Weight * 0.8 // 衰减因子
                
                queue = append(queue, Node{
                    MemoryID:  neighbor,
                    Depth:     current.Depth + 1,
                    Relevance: relevance,
                })
                
                resultEdges = append(resultEdges, edges[i])
            }
        }
    }
    
    return resultNodes, resultEdges, nil
}

// DFSSearch 深度优先搜索
func (g *Graph) DFSSearch(
    ctx context.Context,
    startID string,
    maxDepth int,
    maxResults int,
    edgeTypes []string,
    minWeight float32,
) ([]Node, []Edge, error) {
    
    visited := make(map[string]bool)
    var resultNodes []Node
    var resultEdges []Edge
    
    var dfs func(nodeID string, depth int, relevance float32) error
    dfs = func(nodeID string, depth int, relevance float32) error {
        // 检查终止条件
        if visited[nodeID] || depth > maxDepth || len(resultNodes) >= maxResults {
            return nil
        }
        
        visited[nodeID] = true
        resultNodes = append(resultNodes, Node{
            MemoryID:  nodeID,
            Depth:     depth,
            Relevance: relevance,
        })
        
        // 查询邻居
        neighbors, edges, err := g.getNeighbors(ctx, nodeID, edgeTypes, minWeight)
        if err != nil {
            return err
        }
        
        // 递归访问邻居
        for i, neighbor := range neighbors {
            if !visited[neighbor] {
                resultEdges = append(resultEdges, edges[i])
                newRelevance := relevance * edges[i].Weight * 0.8
                if err := dfs(neighbor, depth+1, newRelevance); err != nil {
                    return err
                }
            }
        }
        
        return nil
    }
    
    if err := dfs(startID, 0, 1.0); err != nil {
        return nil, nil, err
    }
    
    return resultNodes, resultEdges, nil
}

// BidirectionalSearch 双向搜索（起点和终点同时搜索）
func (g *Graph) BidirectionalSearch(
    ctx context.Context,
    startID string,
    targetID string,
    maxDepth int,
) ([]Node, []Edge, error) {
    
    // 从起点开始的访问集合
    forwardVisited := make(map[string]*Node)
    forwardVisited[startID] = &Node{MemoryID: startID, Depth: 0, Relevance: 1.0}
    forwardQueue := []string{startID}
    
    // 从终点开始的访问集合
    backwardVisited := make(map[string]*Node)
    backwardVisited[targetID] = &Node{MemoryID: targetID, Depth: 0, Relevance: 1.0}
    backwardQueue := []string{targetID}
    
    var resultEdges []Edge
    
    for depth := 0; depth <= maxDepth/2; depth++ {
        // 正向扩展一层
        newForwardQueue := []string{}
        for _, nodeID := range forwardQueue {
            neighbors, edges, err := g.getNeighbors(ctx, nodeID, nil, 0.0)
            if err != nil {
                return nil, nil, err
            }
            
            for i, neighbor := range neighbors {
                // 检查是否与反向搜索相遇
                if backwardVisited[neighbor] != nil {
                    // 找到路径！
                    return g.reconstructPath(forwardVisited, backwardVisited, neighbor, resultEdges)
                }
                
                if forwardVisited[neighbor] == nil {
                    forwardVisited[neighbor] = &Node{
                        MemoryID:  neighbor,
                        Depth:     depth + 1,
                        Relevance: forwardVisited[nodeID].Relevance * edges[i].Weight * 0.8,
                    }
                    newForwardQueue = append(newForwardQueue, neighbor)
                    resultEdges = append(resultEdges, edges[i])
                }
            }
        }
        forwardQueue = newForwardQueue
        
        // 反向扩展一层
        newBackwardQueue := []string{}
        for _, nodeID := range backwardQueue {
            // 查询反向边（incoming edges）
            neighbors, edges, err := g.getIncomingNeighbors(ctx, nodeID, nil, 0.0)
            if err != nil {
                return nil, nil, err
            }
            
            for i, neighbor := range neighbors {
                // 检查是否与正向搜索相遇
                if forwardVisited[neighbor] != nil {
                    return g.reconstructPath(forwardVisited, backwardVisited, neighbor, resultEdges)
                }
                
                if backwardVisited[neighbor] == nil {
                    backwardVisited[neighbor] = &Node{
                        MemoryID:  neighbor,
                        Depth:     depth + 1,
                        Relevance: backwardVisited[nodeID].Relevance * edges[i].Weight * 0.8,
                    }
                    newBackwardQueue = append(newBackwardQueue, neighbor)
                    resultEdges = append(resultEdges, edges[i])
                }
            }
        }
        backwardQueue = newBackwardQueue
        
        // 如果队列都空了，说明没有路径
        if len(forwardQueue) == 0 && len(backwardQueue) == 0 {
            break
        }
    }
    
    // 未找到路径
    return nil, nil, fmt.Errorf("在最大深度内未找到路径")
}

// getNeighbors 查询出边邻居
func (g *Graph) getNeighbors(
    ctx context.Context,
    nodeID string,
    edgeTypes []string,
    minWeight float32,
) ([]string, []Edge, error) {
    
    query := `
        SELECT target_id, edge_type, weight
        FROM relationships
        WHERE source_id = ?
        AND weight >= ?
    `
    
    args := []interface{}{nodeID, minWeight}
    
    // 添加边类型过滤
    if len(edgeTypes) > 0 {
        placeholders := ""
        for i, edgeType := range edgeTypes {
            if i > 0 {
                placeholders += ", "
            }
            placeholders += "?"
            args = append(args, edgeType)
        }
        query += fmt.Sprintf(" AND edge_type IN (%s)", placeholders)
    }
    
    rows, err := g.db.QueryContext(ctx, query, args...)
    if err != nil {
        return nil, nil, err
    }
    defer rows.Close()
    
    var neighbors []string
    var edges []Edge
    
    for rows.Next() {
        var targetID, edgeType string
        var weight float32
        
        if err := rows.Scan(&targetID, &edgeType, &weight); err != nil {
            return nil, nil, err
        }
        
        neighbors = append(neighbors, targetID)
        edges = append(edges, Edge{
            SourceID: nodeID,
            TargetID: targetID,
            EdgeType: edgeType,
            Weight:   weight,
        })
    }
    
    return neighbors, edges, nil
}

// reconstructPath 重建路径（双向搜索用）
func (g *Graph) reconstructPath(
    forwardVisited map[string]*Node,
    backwardVisited map[string]*Node,
    meetingPoint string,
    edges []Edge,
) ([]Node, []Edge, error) {
    
    var nodes []Node
    
    // 从起点到相遇点
    current := meetingPoint
    for forwardVisited[current] != nil {
        nodes = append([]Node{*forwardVisited[current]}, nodes...)
        // 这里需要存储父节点信息，简化起见省略
        break
    }
    
    // 从相遇点到终点
    current = meetingPoint
    for backwardVisited[current] != nil {
        nodes = append(nodes, *backwardVisited[current])
        break
    }
    
    return nodes, edges, nil
}

// ========== A* 搜索（启发式搜索） ==========

// AStarSearch A* 搜索（适合有明确目标的搜索）
func (g *Graph) AStarSearch(
    ctx context.Context,
    startID string,
    targetID string,
    maxDepth int,
) ([]Node, []Edge, error) {
    
    // 优先队列（按 f = g + h 排序）
    pq := &PriorityQueue{}
    heap.Init(pq)
    
    // g: 起点到当前点的实际代价
    // h: 当前点到终点的启发式估计
    // f: g + h
    
    startItem := &Item{
        node: Node{
            MemoryID:  startID,
            Depth:     0,
            Relevance: 1.0,
        },
        priority: 0, // f值
        gScore:   0, // 实际代价
    }
    heap.Push(pq, startItem)
    
    visited := make(map[string]bool)
    gScores := make(map[string]float32)
    gScores[startID] = 0
    
    var resultNodes []Node
    var resultEdges []Edge
    
    for pq.Len() > 0 {
        // 取出 f 值最小的节点
        current := heap.Pop(pq).(*Item)
        currentID := current.node.MemoryID
        
        // 如果到达目标
        if currentID == targetID {
            resultNodes = append(resultNodes, current.node)
            return resultNodes, resultEdges, nil
        }
        
        // 如果已访问
        if visited[currentID] {
            continue
        }
        visited[currentID] = true
        resultNodes = append(resultNodes, current.node)
        
        // 如果达到最大深度
        if current.node.Depth >= maxDepth {
            continue
        }
        
        // 扩展邻居
        neighbors, edges, err := g.getNeighbors(ctx, currentID, nil, 0.0)
        if err != nil {
            return nil, nil, err
        }
        
        for i, neighbor := range neighbors {
            if visited[neighbor] {
                continue
            }
            
            // 计算 g 值（起点到邻居的代价）
            tentativeG := gScores[currentID] + (1.0 - edges[i].Weight)
            
            // 如果找到更好的路径
            if existingG, exists := gScores[neighbor]; !exists || tentativeG < existingG {
                gScores[neighbor] = tentativeG
                
                // 计算 h 值（启发式估计：这里简化为固定值）
                h := g.heuristic(neighbor, targetID)
                
                // f = g + h
                f := tentativeG + h
                
                heap.Push(pq, &Item{
                    node: Node{
                        MemoryID:  neighbor,
                        Depth:     current.node.Depth + 1,
                        Relevance: current.node.Relevance * edges[i].Weight * 0.8,
                    },
                    priority: f,
                    gScore:   tentativeG,
                })
                
                resultEdges = append(resultEdges, edges[i])
            }
        }
    }
    
    return resultNodes, resultEdges, fmt.Errorf("未找到路径")
}

// heuristic 启发式函数（简化版，实际可以用向量距离等）
func (g *Graph) heuristic(fromID, toID string) float32 {
    // 这里简化为固定值
    // 实际可以查询两个节点的向量，计算余弦距离
    return 1.0
}

// ========== 优先队列实现（A* 用） ==========

type Item struct {
    node     Node
    priority float32 // f值
    gScore   float32
    index    int
}

type PriorityQueue []*Item

func (pq PriorityQueue) Len() int { return len(pq) }

func (pq PriorityQueue) Less(i, j int) bool {
    return pq[i].priority < pq[j].priority
}

func (pq PriorityQueue) Swap(i, j int) {
    pq[i], pq[j] = pq[j], pq[i]
    pq[i].index = i
    pq[j].index = j
}

func (pq *PriorityQueue) Push(x interface{}) {
    n := len(*pq)
    item := x.(*Item)
    item.index = n
    *pq = append(*pq, item)
}

func (pq *PriorityQueue) Pop() interface{} {
    old := *pq
    n := len(old)
    item := old[n-1]
    old[n-1] = nil
    item.index = -1
    *pq = old[0 : n-1]
    return item
}
```

```go
    ctx context.Context,

            ctx,

            int(req.MaxDepth),

            req.EdgeTypes,
            req.MinWeight,

// HybridSearch 实现混合搜索
func (s *SearchService) HybridSearch(
    ctx context.Context,
    req *pb.HybridSearchRequest,
) (*pb.HybridSearchResponse, error) {
    
    startTime := time.Now()
    
    // 1. 全文搜索
    fulltextResults, err := s.searchIndex.FulltextSearch(
        ctx,
        req.SearchReq.Query,
        int(req.SearchReq.Limit),
    )
    if err != nil {
        return nil, err
    }
    
    // 2. 图谱搜索（如果提供了起始节点）
    var graphResults []string
    if req.GraphReq != nil && req.GraphReq.StartMemoryId != "" {
        nodes, _, err := s.graph.BFSSearch(
            ctx,
            req.GraphReq.StartMemoryId,
            int(req.GraphReq.MaxDepth),
            int(req.GraphReq.MaxResults),
            req.GraphReq.EdgeTypes,
            req.GraphReq.MinWeight,
        )
        if err == nil {
            for _, node := range nodes {
                graphResults = append(graphResults, node.MemoryID)
            }
        }
    }
    
    // 3. 结果融合
    merged := s.mergeResults(
        fulltextResults,
        graphResults,
        req.Weights,
    )
    
    elapsed := float32(time.Since(startTime).Microseconds()) / 1000.0
    s.stats.RecordSearch("hybrid", elapsed)
    
    return &pb.HybridSearchResponse{
        MemoryIds:     merged.IDs,
        ScoreBreakdown: merged.Breakdown,
        SearchTimeMs:  elapsed,
    }, nil
}

// mergeResults 融合多路搜索结果
func (s *SearchService) mergeResults(
    fulltextResults []search.Result,
    graphResults []string,
    weights *pb.HybridWeights,
) *MergedResults {
    
    // 评分映射
    scores := make(map[string]*pb.ScoreBreakdown)
    
    // 全文搜索评分
    for i, result := range fulltextResults {
        score := float32(len(fulltextResults)-i) / float32(len(fulltextResults))
        scores[result.MemoryID] = &pb.ScoreBreakdown{
            FulltextScore: score * weights.FulltextWeight,
            SemanticScore: 0,
            GraphScore:    0,
        }
    }
    
    // 图谱搜索评分
    for i, memoryID := range graphResults {
        score := float32(len(graphResults)-i) / float32(len(graphResults))
        
        if existing, ok := scores[memoryID]; ok {
            existing.GraphScore = score * weights.GraphWeight
        } else {
            scores[memoryID] = &pb.ScoreBreakdown{
                FulltextScore: 0,
                SemanticScore: 0,
                GraphScore:    score * weights.GraphWeight,
            }
        }
    }
    
    // 计算最终评分并排序
    type scoredID struct {
        id    string
        score float32
    }
    
    var sorted []scoredID
    for id, breakdown := range scores {
        breakdown.FinalScore = breakdown.FulltextScore +
            breakdown.SemanticScore +
            breakdown.GraphScore
        
        sorted = append(sorted, scoredID{id, breakdown.FinalScore})
    }
    
    // 排序
    sort.Slice(sorted, func(i, j int) bool {
        return sorted[i].score > sorted[j].score
    })
    
    // 提取结果
    var ids []string
    for _, item := range sorted {
        ids = append(ids, item.id)
        if len(ids) >= 20 {
            break
        }
    }
    
    return &MergedResults{
        IDs:       ids,
        Breakdown: scores,
    }
}

type MergedResults struct {
    IDs       []string
    Breakdown map[string]*pb.ScoreBreakdown
}

// Stats 统计信息

    totalSearchTime float64
    mu              sync.RWMutex
}

func NewStats() *Stats {
    return &Stats{}
}

func (s *Stats) RecordSearch(searchType string, timeMs float32) {
    s.mu.Lock()
    defer s.mu.Unlock()
    
    s.totalSearches++
    s.totalSearchTime += float64(timeMs)
}

func (s *Stats) GetAvgSearchTime() float32 {
    s.mu.RLock()
    defer s.mu.RUnlock()
    
    if s.totalSearches == 0 {
        return 0
    }
    
    return float32(s.totalSearchTime / float64(s.totalSearches))
}
```

```bash
        -I. \
        --python_out=python/deepseek_memory \
        --grpc_python_out=python/deepseek_memory \
        proto/memory.proto
    
    # 修复 Python 导入
    if [[ "$OSTYPE" == "darwin"* ]]; then
        sed -i '' 's/import memory_pb2/from . import memory_pb2/g' python/deepseek_memory/memory_pb2_grpc.py
    else
        sed -i 's/import memory_pb2/from . import memory_pb2/g' python/deepseek_memory/memory_pb2_grpc.py
    fi
    echo "✓ Python protobuf 代码已生成"
    
    echo ""
}

# 构建 Go 服务（多平台）
build_go() {
    echo -e "${YELLOW}📦 构建 Go 服务...${NC}"
    
    cd go-service
    
    # 定义目标平台
    platforms=(
        "linux/amd64"
        "linux/arm64"
        "darwin/amd64"
        "darwin/arm64"
        "windows/amd64"
    )
    
    for platform in "${platforms[@]}"; do
        IFS='/' read -r goos goarch <<< "$platform"
        
        output="../binaries/${goos}-${goarch}/memory-service"
        if [ "$goos" = "windows" ]; then
            output="${output}.exe"
        fi
        
        echo "  构建 ${goos}/${goarch}..."
        
        GOOS=$goos GOARCH=$goarch CGO_ENABLED=0 \
            go build -ldflags="-s -w" -o "$output" cmd/memory-service/main.go
        
        if [ $? -eq 0 ]; then
            echo -e "  ${GREEN}✓${NC} ${goos}/${goarch}"
        else
            echo -e "  ${RED}✗${NC} ${goos}/${goarch} 构建失败"
            cd ..
            return 1
        fi
    done
    
    cd ..
    echo ""
}

# 构建 Rust 库（多平台）
build_rust() {
    echo -e "${YELLOW}🦀 构建 Rust 库...${NC}"
    
    cd rust-core
    
    # 检查是否安装了 cross（用于交叉编译）
    if ! command -v cross &> /dev/null; then
        echo "  安装 cross 工具..."
        cargo install cross --git https://github.com/cross-rs/cross
    fi
    
    # 定义目标平台
    declare -A targets=(
        ["linux-amd64"]="x86_64-unknown-linux-gnu"
        ["linux-arm64"]="aarch64-unknown-linux-gnu"
        ["darwin-amd64"]="x86_64-apple-darwin"
        ["darwin-arm64"]="aarch64-apple-darwin"
        ["windows-amd64"]="x86_64-pc-windows-msvc"
    )
    
    declare -A lib_names=(
        ["linux-amd64"]="libmemory_core.so"
        ["linux-arm64"]="libmemory_core.so"
        ["darwin-amd64"]="libmemory_core.dylib"
        ["darwin-arm64"]="libmemory_core.dylib"
        ["windows-amd64"]="memory_core.dll"
    )
    
    for platform in "${!targets[@]}"; do
        target="${targets[$platform]}"
        lib_name="${lib_names[$platform]}"
        
        echo "  构建 ${platform} (${target})..."
        
        # 添加目标
        rustup target add "$target" 2>/dev/null || true
        
        # 根据平台选择构建方式
        if [[ "$platform" == "linux-arm64" ]] && [[ "$OSTYPE" != "linux-gnu"* ]]; then
            # 交叉编译 ARM Linux
            cross build --release --target "$target"
        else
            cargo build --release --target "$target"
        fi
        
        if [ $? -eq 0 ]; then
            # 复制到 binaries 目录
            mkdir -p "../binaries/${platform}"
            cp "target/${target}/release/${lib_name}" "../binaries/${platform}/"
            echo -e "  ${GREEN}✓${NC} ${platform}"
        else
            echo -e "  ${RED}✗${NC} ${platform} 构建失败"
            cd ..
            return 1
        fi
    done
    
    cd ..
    echo ""
}

# 打包 Python wheel
package_python() {
    echo -e "${YELLOW}🐍 打包 Python wheel...${NC}"
    
    # 复制二进制到 Python 包
    echo "  复制二进制文件..."
    rm -rf python/deepseek_memory/binaries
    cp -r binaries python/deepseek_memory/
    
    # 设置执行权限
    find python/deepseek_memory/binaries -name "memory-service*" -exec chmod +x {} \;
    
    cd python
    
    # 安装构建工具
    pip install build twine --quiet
    
    # 清理旧的构建产物
    rm -rf dist/ build/ *.egg-info
    
    # 构建
    echo "  构建 wheel..."
    python3 -m build
    
    if [ $? -eq 0 ]; then
        echo -e "${GREEN}✓ Wheel 构建成功${NC}"
        echo "  输出: $(ls dist/*.whl)"
    else
        echo -e "${RED}✗ Wheel 构建失败${NC}"
        cd ..
        return 1
    fi
    
    cd ..
    echo ""
}
```
