### 1. **数据迁移与版本兼容** ⚠️ 非常重要

用户升级插件时，数据库 schema 可能变化，需要自动迁移。

```python
# python/deepseek_memory/migration/migrator.py
"""
数据库迁移系统
"""
import sqlite3
from pathlib import Path
from typing import List, Callable
import logging

logger = logging.getLogger(__name__)


class Migration:
    """单个迁移任务"""
    
    def __init__(self, version: int, description: str, upgrade: Callable, downgrade: Callable):
        self.version = version
        self.description = description
        self.upgrade = upgrade
        self.downgrade = downgrade


class Migrator:
    """迁移管理器"""
    
    def __init__(self, db_path: Path):
        self.db_path = db_path
        self.conn = sqlite3.connect(str(db_path))
        self.migrations: List[Migration] = []
        
        # 初始化迁移表
        self._init_migration_table()
        
        # 注册所有迁移
        self._register_migrations()
    
    def _init_migration_table(self):
        """创建迁移记录表"""
        self.conn.execute("""
            CREATE TABLE IF NOT EXISTS schema_migrations (
                version INTEGER PRIMARY KEY,
                description TEXT NOT NULL,
                applied_at REAL NOT NULL
            )
        """)
        self.conn.commit()
    
    def _register_migrations(self):
        """注册所有迁移（按版本顺序）"""
        
        # Migration 1: 初始 schema
        self.migrations.append(Migration(
            version=1,
            description="初始数据库结构",
            upgrade=self._migration_001_upgrade,
            downgrade=self._migration_001_downgrade
        ))
        
        # Migration 2: 添加 importance_score 字段
        self.migrations.append(Migration(
            version=2,
            description="添加重要性评分字段",
            upgrade=self._migration_002_upgrade,
            downgrade=self._migration_002_downgrade
        ))
        
        # Migration 3: 添加全文搜索索引
        self.migrations.append(Migration(
            version=3,
            description="添加 FTS5 全文搜索",
            upgrade=self._migration_003_upgrade,
            downgrade=self._migration_003_downgrade
        ))
        
        # 后续迁移在这里添加...
    
    def get_current_version(self) -> int:
        """获取当前数据库版本"""
        cursor = self.conn.execute("""
            SELECT MAX(version) FROM schema_migrations
        """)
        result = cursor.fetchone()[0]
        return result if result else 0
    
    def get_target_version(self) -> int:
        """获取目标版本（最新迁移版本）"""
        return max(m.version for m in self.migrations) if self.migrations else 0
    
    def migrate(self):
        """执行迁移"""
        current = self.get_current_version()
        target = self.get_target_version()
        
        if current == target:
            logger.info(f"数据库已是最新版本: v{current}")
            return
        
        if current > target:
            raise ValueError(f"数据库版本 v{current} 高于代码版本 v{target}，请升级插件")
        
        logger.info(f"开始迁移: v{current} → v{target}")
        
        # 执行每个迁移
        for migration in self.migrations:
            if migration.version > current:
                logger.info(f"应用迁移 v{migration.version}: {migration.description}")
                
                try:
                    # 开启事务
                    self.conn.execute("BEGIN")
                    
                    # 执行升级
                    migration.upgrade(self.conn)
                    
                    # 记录迁移
                    import time
                    self.conn.execute("""
                        INSERT INTO schema_migrations (version, description, applied_at)
                        VALUES (?, ?, ?)
                    """, (migration.version, migration.description, time.time()))
                    
                    # 提交事务
                    self.conn.commit()
                    
                    logger.info(f"✓ 迁移 v{migration.version} 完成")
                
                except Exception as e:
                    # 回滚
                    self.conn.rollback()
                    logger.error(f"✗ 迁移 v{migration.version} 失败: {e}")
                    raise
        
        logger.info(f"✓ 迁移完成，当前版本: v{self.get_current_version()}")
    
    # ========== 具体迁移实现 ==========
    
    def _migration_001_upgrade(self, conn):
        """v1: 初始 schema"""
        conn.execute("""
            CREATE TABLE IF NOT EXISTS memories (
                id TEXT PRIMARY KEY,
                layer TEXT NOT NULL,
                content TEXT NOT NULL,
                created_at REAL NOT NULL
            )
        """)
    
    def _migration_001_downgrade(self, conn):
        """回滚 v1"""
        conn.execute("DROP TABLE IF EXISTS memories")
    
    def _migration_002_upgrade(self, conn):
        """v2: 添加 importance_score"""
        conn.execute("""
            ALTER TABLE memories ADD COLUMN importance_score REAL DEFAULT 0.5
        """)
        conn.execute("""
            ALTER TABLE memories ADD COLUMN access_count INTEGER DEFAULT 0
        """)
    
    def _migration_002_downgrade(self, conn):
        """回滚 v2（SQLite 不支持 DROP COLUMN，需要重建表）"""
        conn.execute("""
            CREATE TABLE memories_backup AS 
            SELECT id, layer, content, created_at FROM memories
        """)
        conn.execute("DROP TABLE memories")
        conn.execute("""
            ALTER TABLE memories_backup RENAME TO memories
        """)
    
    def _migration_003_upgrade(self, conn):
        """v3: 添加 FTS5"""
        conn.execute("""
            CREATE VIRTUAL TABLE IF NOT EXISTS memories_fts
            USING fts5(id, content, tokenize='porter unicode61')
        """)
        # 迁移现有数据
        conn.execute("""
            INSERT INTO memories_fts (id, content)
            SELECT id, content FROM memories
        """)
    
    def _migration_003_downgrade(self, conn):
        """回滚 v3"""
        conn.execute("DROP TABLE IF EXISTS memories_fts")
```

### 3. **错误恢复与数据备份** ⚠️ 非常重要

Go 服务崩溃、数据损坏时的恢复机制。

```python
# python/deepseek_memory/recovery.py
"""
错误恢复与数据备份
"""
import shutil
import sqlite3
from pathlib import Path
from datetime import datetime
import logging

logger = logging.getLogger(__name__)


class RecoveryManager:
    """恢复管理器"""
    
    def __init__(self, data_dir: Path):
        self.data_dir = data_dir
        self.backup_dir = data_dir / "backups"
        self.backup_dir.mkdir(exist_ok=True)
    
    def create_backup(self, tag: str = None) -> Path:
        """
        创建备份
        
        Args:
            tag: 备份标签（如 "before_upgrade"）
        
        Returns:
            备份文件路径
        """
        timestamp = datetime.now().strftime("%Y%m%d_%H%M%S")
        backup_name = f"backup_{timestamp}"
        if tag:
            backup_name += f"_{tag}"
        backup_name += ".db"
        
        source = self.data_dir / "data.db"
        target = self.backup_dir / backup_name
        
        logger.info(f"创建备份: {backup_name}")
        
        # 使用 SQLite 的 backup API（保证一致性）
        source_conn = sqlite3.connect(str(source))
        target_conn = sqlite3.connect(str(target))
        
        with target_conn:
            source_conn.backup(target_conn)
        
        source_conn.close()
        target_conn.close()
        
        logger.info(f"✓ 备份完成: {target}")
        return target
    
    def restore_backup(self, backup_path: Path):
        """恢复备份"""
        db_path = self.data_dir / "data.db"
        
        logger.warning(f"恢复备份: {backup_path} → {db_path}")
        
        # 先备份当前损坏的数据库
        if db_path.exists():
            corrupt_backup = self.data_dir / f"data_corrupt_{datetime.now().strftime('%Y%m%d_%H%M%S')}.db"
            shutil.move(str(db_path), str(corrupt_backup))
            logger.info(f"损坏的数据库已移动到: {corrupt_backup}")
        
        # 恢复
        shutil.copy2(str(backup_path), str(db_path))
        logger.info(f"✓ 备份恢复完成")
    
    def list_backups(self) -> List[Path]:
        """列出所有备份"""
        return sorted(self.backup_dir.glob("backup_*.db"), reverse=True)
    
    def cleanup_old_backups(self, keep_last: int = 10):
        """清理旧备份"""
        backups = self.list_backups()
        
        if len(backups) > keep_last:
            to_delete = backups[keep_last:]
            for backup in to_delete:
                backup.unlink()
                logger.info(f"删除旧备份: {backup.name}")
    
    def verify_database(self) -> bool:
        """验证数据库完整性"""
        db_path = self.data_dir / "data.db"
        
        try:
            conn = sqlite3.connect(str(db_path))
            cursor = conn.execute("PRAGMA integrity_check")
            result = cursor.fetchone()[0]
            conn.close()
            
            if result == "ok":
                logger.info("✓ 数据库完整性检查通过")
                return True
            else:
                logger.error(f"✗ 数据库损坏: {result}")
                return False
        
        except Exception as e:
            logger.error(f"✗ 数据库验证失败: {e}")
            return False
    
    def auto_recover(self) -> bool:
        """
        自动恢复（如果数据库损坏）
        
        Returns:
            是否成功恢复
        """
        if self.verify_database():
            return True
        
        logger.warning("检测到数据库损坏，尝试自动恢复...")
        
        backups = self.list_backups()
        if not backups:
            logger.error("✗ 没有可用的备份，无法恢复")
            return False
        
        # 尝试最近的备份
        for backup in backups[:3]:  # 最多尝试 3 个最近的备份
            logger.info(f"尝试恢复备份: {backup.name}")
            
            self.restore_backup(backup)
            
            if self.verify_database():
                logger.info(f"✓ 成功从备份恢复: {backup.name}")
                return True
            else:
                logger.warning(f"✗ 备份 {backup.name} 也已损坏，尝试下一个...")
        
        logger.error("✗ 所有备份都无法恢复")
        return False


# 集成到 MemoryManager
class MemoryManager:
    def __init__(self, config=None, data_dir=None):
        # ... 其他初始化
        
        # 初始化恢复管理器
        self.recovery = RecoveryManager(self.data_dir)
        
        # 启动时验证数据库
        if not self.recovery.verify_database():
            logger.warning("数据库损坏，尝试自动恢复...")
            if not self.recovery.auto_recover():
                raise RuntimeError("数据库损坏且无法恢复，请联系支持")
        
        # 定期自动备份（每天一次）
        if config.maintenance.auto_backup:
            from apscheduler.schedulers.background import BackgroundScheduler
            scheduler = BackgroundScheduler()
            scheduler.add_job(
                func=lambda: self.recovery.create_backup("auto"),
                trigger='cron',
                hour=4,  # 凌晨4点
                id='auto_backup'
            )
            scheduler.start()
```

```python
# python/deepseek_memory/limits.py
"""
资源限制管理
"""
import psutil
import os
from pathlib import Path
import logging

logger = logging.getLogger(__name__)


class ResourceLimiter:
    """资源限制器"""
    
    def __init__(self, config: dict):
        self.max_memory_mb = config.get('max_memory_mb', 1024)  # 默认 1GB
        self.max_disk_mb = config.get('max_disk_mb', 10240)    # 默认 10GB
        self.max_memories = config.get('max_memories', 1000000) # 默认 100万条
        
        self.warning_threshold = 0.8  # 达到 80% 时警告
    
    def check_memory_usage(self) -> dict:
        """检查内存使用"""
        process = psutil.Process(os.getpid())
        memory_mb = process.memory_info().rss / 1024 / 1024
        
        usage_ratio = memory_mb / self.max_memory_mb
        
        return {
            'current_mb': memory_mb,
            'limit_mb': self.max_memory_mb,
            'usage_ratio': usage_ratio,
            'warning': usage_ratio > self.warning_threshold
        }
    
    def check_disk_usage(self, data_dir: Path) -> dict:
        """检查磁盘使用"""
        total_size = 0
        for file in data_dir.rglob('*'):
            if file.is_file():
                total_size += file.stat().st_size
        
        size_mb = total_size / 1024 / 1024
        usage_ratio = size_mb / self.max_disk_mb
        
        return {
            'current_mb': size_mb,
            'limit_mb': self.max_disk_mb,
            'usage_ratio': usage_ratio,
            'warning': usage_ratio > self.warning_threshold
        }
    
    def enforce_limits(self, storage) -> dict:
        """
        强制执行限制
        
        Returns:
            清理报告
        """
        stats = storage.get_stats()
        
        actions = {
            'cleaned_memories': 0,
            'cleaned_events': 0,
            'reason': []
        }
        
        # 检查记忆数量
        if stats['total_memories'] > self.max_memories:
            excess = stats['total_memories'] - int(self.max_memories * 0.9)  # 清理到 90%
            logger.warning(f"记忆数量超限（{stats['total_memories']} > {self.max_memories}），清理最旧的 {excess} 条")
            
            # 删除最旧的轮转记忆
            cleaned = self._cleanup_oldest_memories(storage, 'L6', excess)
            actions['cleaned_memories'] = cleaned
            actions['reason'].append(f"记忆数量超限")
        
        # 检查磁盘使用
        disk = self.check_disk_usage(Path(storage.db_path).parent)
        if disk['warning']:
            logger.warning(f"磁盘使用超限（{disk['usage_ratio']:.1%}）")
            
            # 清理旧事件
            cleaned_events = self._cleanup_old_events(storage, keep_days=30)
            actions['cleaned_events'] = cleaned_events
            actions['reason'].append(f"磁盘使用超限（{disk['usage_ratio']:.1%}）")
        
        return actions
    
    def _cleanup_oldest_memories(self, storage, layer: str, count: int) -> int:
        """清理最旧的记忆"""
        # 查询最旧的记忆
        query = """
            SELECT id FROM memories
            WHERE layer = ? AND status = 'active'
            ORDER BY created_at ASC
            LIMIT ?
        """
        
        cursor = storage.conn.execute(query, (layer, count))
        ids = [row[0] for row in cursor.fetchall()]
        
        # 删除
        for memory_id in ids:
            storage.delete_memory(memory_id)
        
        return len(ids)
    
    def _cleanup_old_events(self, storage, keep_days: int) -> int:
        """清理旧事件"""
        import time
        cutoff = time.time() - (keep_days * 86400)
        
        cursor = storage.conn.execute("""
            DELETE FROM events WHERE timestamp < ?
        """, (cutoff,))
        
        storage.conn.commit()
        return cursor.rowcount
```

```python
# python/deepseek_memory/process_manager.py (增强版)
import subprocess
import time
import threading
import logging

logger = logging.getLogger(__name__)


class GoServiceManager:
    def __init__(self, data_dir, port=None):
        # ... 原有代码
        
        self.restart_count = 0
        self.max_restarts = 5
        self.restart_window = 60  # 60秒内最多重启5次
        self.last_restart_time = 0
        
        # 启动监控线程
        self.monitor_thread = threading.Thread(target=self._monitor_loop, daemon=True)
        self.monitor_thread.start()
    
    def _monitor_loop(self):
        """监控循环（后台线程）"""
        while True:
            time.sleep(5)  # 每5秒检查一次
            
            if not self.is_running():
                logger.warning("⚠️ Go 服务已停止，尝试重启...")
                
                # 检查重启频率
                now = time.time()
                if now - self.last_restart_time > self.restart_window:
                    # 重置计数器
                    self.restart_count = 0
                
                if self.restart_count >= self.max_restarts:
                    logger.error(f"✗ Go 服务在 {self.restart_window}秒内重启 {self.max_restarts} 次，停止重启")
                    break
                
                # 尝试重启
                try:
                    self.start()
                    self.restart_count += 1
                    self.last_restart_time = now
                    logger.info(f"✓ Go 服务已重启（第 {self.restart_count} 次）")
                
                except Exception as e:
                    logger.error(f"✗ Go 服务重启失败: {e}")
    
    def get_health(self) -> dict:
        """获取健康状态"""
        if not self.is_running():
            return {'healthy': False, 'reason': 'process_not_running'}
        
        try:
            # 调用 gRPC 健康检查
            from .core.search_client import SearchClient
            client = SearchClient(self.get_address())
            health = client.health_check(detailed=True)
            
            return health
        
        except Exception as e:
            return {'healthy': False, 'reason': str(e)}
```

```python
        """初始化用户表"""

            CREATE TABLE IF NOT EXISTS users (
                user_id TEXT PRIMARY KEY,
                username TEXT UNIQUE NOT NULL,
                password_hash TEXT NOT NULL,
                role TEXT NOT NULL,  -- admin/user/readonly
                created_at REAL NOT NULL
            )
        """)
        
        # 为记忆表添加 user_id 字段
        try:
            self.storage.conn.execute("""
                ALTER TABLE memories ADD COLUMN user_id TEXT
            """)
        except:
            pass  # 字段已存在
        
        self.storage.conn.commit()
    
    def create_user(self, username: str, password: str, role: str = 'user') -> str:
        """创建用户"""
        import time
        from ..utils import generate_id
        
        user_id = generate_id()
        password_hash = self._hash_password(password)
        
        self.storage.conn.execute("""
            INSERT INTO users (user_id, username, password_hash, role, created_at)
            VALUES (?, ?, ?, ?, ?)
        """, (user_id, username, password_hash, role, time.time()))
        
        self.storage.conn.commit()
        return user_id
    
    def authenticate(self, username: str, password: str) -> Optional[dict]:
        """认证用户"""
        cursor = self.storage.conn.execute("""
            SELECT user_id, username, role FROM users
            WHERE username = ? AND password_hash = ?
        """, (username, self._hash_password(password)))
        
        row = cursor.fetchone()
        if row:
            return {
                'user_id': row[0],
                'username': row[1],
                'role': row[2]
            }
        return None
    
    def _hash_password(self, password: str) -> str:
        """密码哈希"""
        return hashlib.sha256(password.encode()).hexdigest()
```

```python
from deepseek_memory.plugins import MemoryPlugin


class SentimentAnalysisPlugin(MemoryPlugin):
    """
    情感分析插件
    
    自动分析记忆的情感倾向
    """
    
    def on_memory_created(self, memory: dict):
        """记忆创建时触发"""
        # 分析情感
        sentiment = self._analyze_sentiment(memory['content'])
        
        # 添加元数据
        if 'metadata' not in memory:
            memory['metadata'] = {}
        memory['metadata']['sentiment'] = sentiment
        
        print(f"✓ 情感分析: {memory['id']} → {sentiment}")
    
    def _analyze_sentiment(self, text: str) -> str:
        """简单的情感分析（实际可以用模型）"""
        positive_words = ['好', '棒', '喜欢', '优秀', '成功']
        negative_words = ['差', '糟糕', '讨厌', '失败', '错误']
        
        pos_count = sum(1 for word in positive_words if word in text)
        neg_count = sum(1 for word in negative_words if word in text)
        
        if pos_count > neg_count:
            return 'positive'
        elif neg_count > pos_count:
            return 'negative'
        else:
            return 'neutral'
```

```python
"""

import time
```
